public class DocumentValidator {

    private static final Logger logger =
            LoggerFactory.getLogger(DocumentValidator.class);

    public ValidationResult validate(Document doc) {

        try {

            // FIX 1: validation failure should use IllegalArgumentException
            if (doc == null) {
                throw new IllegalArgumentException("Document is null");
            }

            String content = doc.extractContent();

            // FIX 2: check null before isEmpty()
            if (content == null || content.isEmpty()) {
                throw new IllegalArgumentException("Empty content");
            }

            return runValidationRules(content);

        } catch (IllegalArgumentException e) {

            // FIX 3: expected validation failure logged without stack trace
            logger.warn("Validation failed: {}", e.getMessage());

            return ValidationResult.invalid(e.getMessage());

        } catch (Exception e) {

            // FIX 4: unexpected errors should not be swallowed silently
            logger.error("Unexpected validation error", e);

            throw new RuntimeException("Validation processing failed", e);
        }
    }

    public void validateBatch(List<Document> docs) {

        for (Document doc : docs) {

            try {

                ValidationResult r = validate(doc);

                // FIX: null-safe check before calling isValid()
                if (r != null && r.isValid()) {
                    saveResult(r);
                }

            } catch (Exception e) {

                // FIX: log batch processing failure instead of swallowing
                logger.error("Failed to process document in batch", e);
            }
        }
    }
}
